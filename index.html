-- One row per project: Earned Value + RAG status, reused by every report
CREATE OR REPLACE VIEW v_project_health AS
SELECT
    p.project_id,
    p.project_name,
    pf.portfolio_name,
    e.spi,                     -- schedule performance (EV / PV)
    e.cpi,                     -- cost performance     (EV / AC)
    CASE
        WHEN p.status = 'On Hold'          THEN 'Grey'
        WHEN e.spi < 0.85 OR e.cpi < 0.85  THEN 'Red'
        WHEN e.spi < 0.95 OR e.cpi < 0.95  THEN 'Amber'
        ELSE 'Green'
    END AS rag
FROM projects p
JOIN portfolios pf USING (portfolio_id)
LEFT JOIN v_evm_latest e USING (project_id);
-- Is each project improving or getting worse month over month?
WITH trend AS (
    SELECT
        project_id,
        report_month,
        spi,
        lag(spi) OVER w                                              AS prev_spi,
        avg(spi) OVER (w ROWS BETWEEN 2 PRECEDING AND CURRENT ROW)   AS spi_3m_avg,
        row_number() OVER (PARTITION BY project_id ORDER BY report_month DESC) AS rn
    FROM v_evm_monthly
    WINDOW w AS (PARTITION BY project_id ORDER BY report_month)
)
SELECT project_id, prev_spi, spi, round(spi_3m_avg, 3) AS spi_3m_avg,
       CASE WHEN spi < prev_spi AND spi < spi_3m_avg THEN 'Worsening'
            WHEN spi > prev_spi THEN 'Improving'
            ELSE 'Stable' END AS trend
FROM trend
WHERE rn = 1;
-- Catch reporting gaps before they reach leadership
WITH expected AS (
    SELECT p.project_id, gs::DATE AS report_month
    FROM projects p
    CROSS JOIN LATERAL generate_series(
        date_trunc('month', p.start_date) + INTERVAL '1 month',
        (SELECT max(report_month) FROM monthly_progress mp
          WHERE mp.project_id = p.project_id),
        INTERVAL '1 month') gs
)
SELECT 'Missing monthly report' AS rule, e.project_id, e.report_month
FROM expected e
LEFT JOIN monthly_progress mp USING (project_id, report_month)
WHERE mp.project_id IS NULL
UNION ALL
SELECT 'Completed project without end date', project_id, NULL
FROM projects
WHERE status = 'Completed' AND actual_end_date IS NULL;
